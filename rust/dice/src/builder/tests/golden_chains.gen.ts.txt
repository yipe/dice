const out: Record<string, unknown> = {};
for (const [name, build] of Object.entries(chains)) {
  const entry: Record<string, unknown> = {};
  try {
    const r = build();
    try { entry.expr = r.toExpression(); } catch (e) { entry.exprErr = (e as Error).message; }
    try {
      const p = r.toPMF();
      entry.bins = p.support().map((v) => [v, p.pAt(v)]);
    } catch (e) { entry.pmfErr = (e as Error).message; }
  } catch (e) { entry.buildErr = (e as Error).message; }
  out[name] = entry;
}
