  const entry: Record<string, unknown> = {};

    const r = build();
    const isPmf = typeof r.support === "function" && typeof r.toExpression !== "function";
    if (!isPmf) {
      try { entry.expr = r.toExpression(); } catch (e) { entry.exprErr = (e as Error).message; }
      try { entry.mean = r.toPMF().mean(); } catch (e) { entry.meanErr = (e as Error).message; }
    }
    try {
      const p = isPmf ? r : r.toPMF();
      entry.bins = p.toJSON().bins;
    } catch (e) { entry.pmfErr = (e as Error).message; }
  } catch (e) { entry.buildErr = (e as Error).message; }
  out[name] = entry;
}
